########## PART 1: NEW USER SETUP ##########

# 1. Create a new user (home directory + /bin/bash shell)
sudo useradd -m -s /bin/bash forensic01
# Add to sudo group
sudo usermod -aG sudo forensic01
# Set a password
sudo passwd forensic01

# 2. Switch to the new user
su - forensic01

# 3. Create directory and sample log file
mkdir -p /home/forensic01/logs/
cd /home/forensic01/logs/

echo "2026-10-21 08:00:01 INFO  kernel: system boot completed" > sys_events.log

########## PART 2: LOG FILTERING & ANALYSIS ##########

# 1. grep
grep "ERROR" sys_events.log
grep -E "root|user01" sys_events.log

# 2. Piping & redirection (press q to quit less)
grep "ERROR" sys_events.log | tee error_report.txt | less
grep -E "root|user01" sys_events.log | tee -a error_report.txt | less

# 3. head and tail
head -n 5 sys_events.log
tail -n 5 sys_events.log

# 4. cat
cat sys_events.log

########## PART 3: PERMISSIONS AND OWNERSHIP ##########

# 1. Create file and set permissions (rw owner, none for group/others)
touch confidential.txt
chmod 600 confidential.txt
ls -l confidential.txt

# 2. Change owner to root
sudo chown root confidential.txt
ls -l confidential.txt

# 3. Try to read and edit as forensic01 (both are expected to fail)
cat confidential.txt
echo "test entry" >> confidential.txt

########## PART 4: USER MANAGEMENT ##########

# 1. Create temporary users
sudo useradd -m guest1
sudo useradd -m guest2
